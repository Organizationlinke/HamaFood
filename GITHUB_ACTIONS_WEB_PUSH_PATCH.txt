Add this repository secret in GitHub:

HAMA_VAPID_PUBLIC_KEY = <your VAPID public key>

Then add this environment variable to the Build Flutter Web step:

env:
  HAMA_VAPID_PUBLIC_KEY: ${{ secrets.HAMA_VAPID_PUBLIC_KEY }}

And add this argument to flutter build web:

--dart-define="HAMA_VAPID_PUBLIC_KEY=$HAMA_VAPID_PUBLIC_KEY"

Example:

- name: Build Flutter Web
  env:
    SUPABASE_URL: ${{ secrets.SUPABASE_URL }}
    SUPABASE_PUBLISHABLE_KEY: ${{ secrets.SUPABASE_PUBLISHABLE_KEY }}
    HAMA_AUTH_DOMAIN: ${{ secrets.HAMA_AUTH_DOMAIN }}
    HAMA_VAPID_PUBLIC_KEY: ${{ secrets.HAMA_VAPID_PUBLIC_KEY }}
  run: |
    flutter build web \
      --release \
      --base-href="/${{ github.event.repository.name }}/" \
      --dart-define="SUPABASE_URL=$SUPABASE_URL" \
      --dart-define="SUPABASE_PUBLISHABLE_KEY=$SUPABASE_PUBLISHABLE_KEY" \
      --dart-define="HAMA_AUTH_DOMAIN=$HAMA_AUTH_DOMAIN" \
      --dart-define="HAMA_VAPID_PUBLIC_KEY=$HAMA_VAPID_PUBLIC_KEY"

Do NOT put HAMA_VAPID_PRIVATE_KEY in GitHub Pages or Flutter build variables.
Only the Edge Function receives the private key.
